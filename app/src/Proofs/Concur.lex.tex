  name: breaksConcur,

  body: (((((Kernel::concur ours theirs same) =? 1) && (((ours =? 1) && (theirs =? 1)) && (not (same =? 1)))) || (((Kernel::concur ours theirs same) =? 1) && (not ((ours =? 1) && (theirs =? 1))))) || ((((Kernel::concur ours theirs same) =? 2) && (((ours =? 1) && (theirs =? 1)) || ((not (ours =? 1)) && (not (theirs =? 1))))) || (((Kernel::concur ours theirs same) =? 0) && ((ours =? 1) || (theirs =? 1)))))
}

  name: concurHolds,
  parameters: [ours : Nat, theirs : Nat, same : Nat],
